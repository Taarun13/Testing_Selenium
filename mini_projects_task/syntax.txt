1. Get Attribute:
element.get_attribute("attribute_name")
# Example:
src = logo.get_attribute("src")

2. Action Class (Mouse Hover, Drag and Drop):
from selenium.webdriver.common.action_chains import ActionChains
actions = ActionChains(driver)
actions.move_to_element(element).perform()
# Example: actions.drag_and_drop(source, target).perform()

3. Debugging (Manual or With Breakpoints):
print("Debug Step A")
# Optional interactive debugger:
import pdb; pdb.set_trace()

4. Alert Handling:
alert = driver.switch_to.alert
print(alert.text)
alert.accept()       # Confirm OK
alert.dismiss()      # Click Cancel
alert.send_keys("text")  # Only for prompt alerts

5. Take Screenshot:
driver.save_screenshot("screenshot_name.png")
# Example:
driver.save_screenshot("error_popup.png")

6. Script Executor (Scroll/Click):
driver.execute_script("JavaScript code here")
# Scroll Example:
driver.execute_script("window.scrollBy(0, 500);")
# Click with JS:
driver.execute_script("arguments[0].click();", element)

7. Frame Handling:
driver.switch_to.frame("frame_name_or_id")
# OR by index:
driver.switch_to.frame(0)
# To return to main content:
driver.switch_to.default_content()

8.Window/Tab Handling:
windows = driver.window_handles
driver.switch_to.window(windows[1])  # Switch to new tab
driver.close()                        # Close current tab
driver.switch_to.window(windows[0])  # Return to original tab

9. Select Class for Dropdown:
from selenium.webdriver.support.ui import Select
dropdown = Select(driver.find_element(By.ID, "dropdown_id"))
dropdown.select_by_visible_text("Option Text")
dropdown.select_by_value("value")
dropdown.select_by_index(0)
# Get all options:
for option in dropdown.options:
    print(option.text)

10.Explicit Wait:
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
wait = WebDriverWait(driver, 10)
element = wait.until(EC.presence_of_element_located((By.ID, "element_id")))
# You can replace with EC.element_to_be_clickable or EC.visibility_of_element_located
